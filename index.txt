    function drawComparisonChart(canvasId, partnerData, labelPrefix, colorThisMonth, colorLastMonth) {
        const ctx = document.getElementById(canvasId).getContext('2d');
        const labels = Object.keys(partnerData);
        const dataThisMonth = labels.map(label => partnerData[label].thisMonth);
        const dataLastMonth = labels.map(label => partnerData[label].lastMonth);

        chartInstances[canvasId] = new Chart(ctx, {
            type: 'bar',
            data: {
                labels: labels,
                datasets: [{
                    label: 'Tháng này',
                    data: dataThisMonth,
                    backgroundColor: colorThisMonth,
                    barPercentage: 0.5,
                    categoryPercentage: 0.7
                }, {
                    label: 'Tháng trước',
                    data: dataLastMonth,
                    backgroundColor: colorLastMonth,
                    barPercentage: 0.5,
                    categoryPercentage: 0.7
                }]
            },
            options: {
                responsive: true,
                maintainAspectRatio: false,
                scales: {
                    y: { beginAtZero: true }
                },
                plugins: {
                    tooltip: {
                        callbacks: {
                            label: function(context) {
                                let label = context.dataset.label || '';
                                if (label) {
                                    label += ': ';
                                }
                                if (context.parsed.y !== null) {
                                    label += formatter.format(context.parsed.y);
                                }
                                
                                if (context.dataset.label === 'Tháng này') {
                                    const lastMonthValue = context.chart.data.datasets[1].data[context.dataIndex];
                                    if (lastMonthValue > 0) {
                                        const change = ((context.parsed.y - lastMonthValue) / lastMonthValue) * 100;
                                        label += ` (${change >= 0 ? '+' : ''}${change.toFixed(1)}%)`;
                                    } else if (context.parsed.y > 0) {
                                        label += ` (Mới)`;
                                    }
                                }
                                return label;
                            }
                        }
                    }
                }
            }
        });
    }

    function drawRevenueTrendChart(trendData, days) {
        const labels = [];
        const data = [];
        const today = new Date();
        
        for (let i = days - 1; i >= 0; i--) {
            const date = new Date();
            date.setDate(today.getDate() - i);
            
            const day = String(date.getDate()).padStart(2, '0');
            const month = String(date.getMonth() + 1).padStart(2, '0');
            const year = date.getFullYear();
            const dateKey = `${day}/${month}/${year}`;
            
            const displayLabel = `${date.getDate()}/${date.getMonth() + 1}`;
            
            labels.push(displayLabel);
            data.push(trendData[dateKey] || 0);
        }

        const ctx = document.getElementById('revenueTrendChart').getContext('2d');
        if (chartInstances.revenueTrend) {
            chartInstances.revenueTrend.destroy();
        }
        chartInstances.revenueTrend = new Chart(ctx, {
            type: 'line',
            data: {
                labels: labels,
                datasets: [{
                    label: 'Doanh thu',
                    data: data,
                    fill: true,
                    backgroundColor: 'rgba(54, 162, 235, 0.2)',
                    borderColor: 'rgba(54, 162, 235, 1)',
                    tension: 0.1,
                    pointRadius: 2,
                }]
            },
            options: {
                responsive: true,
                maintainAspectRatio: false,
                scales: { y: { beginAtZero: true } },
                plugins: { legend: { display: false } }
            }
        });
    }

    function populateRecentActivities(activities) {

        list.empty();
        if(!activities || activities.length === 0) {

        let html = '';
        activities.forEach(item => {
            const isImport = item.type === 'import';
            const icon = isImport ? '<i class="ph-fill ph-arrow-down-left text-green-500 text-2xl"></i>' : '<i class="ph-fill ph-arrow-up-right text-red-500 text-2xl"></i>';
            const title = isImport ? `Nhập hàng từ ${item.partner}` : `Bán hàng cho ${item.partner}`;
            const valueColor = isImport ? 'text-green-600' : 'text-red-600';

            html += `

                    <div class="p-2 bg-gray-100 rounded-full">${icon}</div>
                    <div class="flex-1 min-w-0">
                        <p class="font-semibold text-sm truncate">${title}</p>

                    </div>
                    <div class="font-bold text-sm ${valueColor} whitespace-nowrap">${formatter.format(item.value)}</div>
                </li>
            `;
        });
        list.html(html);
    }

    function updateComplexImportRow(row) {
        const productGroup = allData.products.find(p => (p.MA_KHO + '-' + p.MA_HANG) === row.data('unique-id'));
        if (!productGroup) return;

        const quantities = JSON.parse(row.attr('data-sku-quantities') || '{}');
        let totalQty = 0;
        const validSkus = [];
        Object.keys(quantities).forEach(skuCode => {

                totalQty += qty;

    function addProductToTransactionTable(productGroup, type, details = {}) {
        const listBody = $(`#${type}-product-list`);
        const uniqueId = productGroup.MA_KHO + '-' + productGroup.MA_HANG;
        if (listBody.find(`tr[data-unique-id="${uniqueId}"]`).length > 0) { showToast('Sản phẩm này đã có trong phiếu.'); return; }
        if (listBody.find('.no-product-row').length > 0) listBody.empty();

        const isComplex = productGroup.skus.length > 1 || (productGroup.skus.length === 1 && productGroup.skus[0].MA_SKU !== productGroup.MA_HANG);
        const productCount = listBody.find('tr.product-row').length + 1;
        
        let tkNo, tkCo, donGia;
        if (type === 'import') {
            tkCo = $('#supplier-tk-co').val() || '';
            tkNo = details.TK_NO || '1561';
            donGia = details.DON_GIA !== undefined ? details.DON_GIA : (productGroup.GIA_NHAP || 0);
        } else { // export
            tkNo = $('#customer-tk-no').val() || '';
            tkCo = details.TK_CO || '5111';
            donGia = details.DON_GIA !== undefined ? details.DON_GIA : (productGroup.GIA_BAN || 0);
        }

        const soLuong = details.SO_LUONG || (isComplex ? 0 : 1);
        const thueSuat = details.THUE_SUAT !== undefined ? details.THUE_SUAT : (parseFloat($(`#${type}StockForm input[name="THUE_SUAT_CHUNG"]`).val()) || 0);
        const thanhTien = soLuong * donGia;
        const tienThue = thanhTien * (thueSuat / 100);

        const skuQuantities = details.skuQuantities ? JSON.stringify(details.skuQuantities) : '{}';

        let newRowHtml = `<tr class="product-row" data-unique-id="${uniqueId}" data-product-type="${isComplex ? 'complex' : 'simple'}" data-sku-quantities='${skuQuantities}'>

                    <div class="font-bold text-gray-800">${productGroup.TEN_HANG}</div>
                    <div class="text-xs text-gray-500">Mã kho: ${productGroup.MA_KHO} | Mã gốc: ${productGroup.MA_HANG}</div>
                    ${isComplex ? '<button type="button" class="open-sku-modal-btn text-xs text-blue-600 font-semibold hover:underline mt-1">Nhập chi tiết SKU</button>' : ''}
                </td>
                <td class="p-2 align-middle">${productGroup.DVT}</td>
                <td class="p-2 align-middle"><input type="number" class="so-luong-input border rounded-md w-full py-1 px-2 ${isComplex ? 'bg-gray-100' : ''}" value="${soLuong}" ${isComplex ? 'readonly' : 'min="0"'}></td>
                <td class="p-2 align-middle"><input type="number" class="don-gia-input border rounded-md w-full py-1 px-2" value="${donGia}"></td>
                <td class="p-2 align-middle"><input type="text" class="thanh-tien-input border rounded-md w-full py-1 px-2 text-right" value="${formatter.format(thanhTien)}"></td>
                <td class="p-2 align-middle"><input type="number" class="thue-suat-input border rounded-md w-full py-1 px-2" value="${thueSuat}"></td>

            </tr>`;

    $('#auto-fill-plan-checkbox').on('change', function() {
        const isChecked = $(this).is(':checked');
        const productGroup = allData.products.find(p => (p.MA_KHO + '-' + p.MA_HANG) === (currentEditingImportRowId || currentEditingExportRowId));
        if (!productGroup) return;

        $('#sku-quantity-list tr').each(function() {
            const row = $(this);
            const skuCode = row.data('sku');
            const skuInfo = productGroup.skus.find(s => s.MA_SKU === skuCode);
            if (!skuInfo) return;

            let qtyToFill = 0;
            if (isChecked) {
                const keHoach = parseFloat(skuInfo.KE_HOACH) || 0;

                if (currentEditingImportRowId) { // Chế độ nhập kho
                    const daNhap = parseFloat(skuInfo.DA_NHAP) || 0;
                    const tongNhapPS = parseFloat(skuInfo.TONG_NHAP_PS) || 0;
                    const luyKeNhap = daNhap + tongNhapPS;
                    qtyToFill = keHoach - luyKeNhap;
                } else { // Chế độ xuất kho
                    const daXuat = parseFloat(skuInfo.DA_XUAT) || 0;
                    const tongXuatPS = parseFloat(skuInfo.TONG_XUAT_PS) || 0;
                    const luyKeXuat = daXuat + tongXuatPS;
                    qtyToFill = keHoach - luyKeXuat;
                    
                    const tonKhoThucTe = parseFloat(skuInfo.TON_KHO_THUC_TE) || 0;
                    // Đảm bảo không xuất quá tồn kho
                    qtyToFill = Math.min(qtyToFill, tonKhoThucTe);
                }

                // Đảm bảo số lượng không âm
                if (qtyToFill < 0) {
                    qtyToFill = 0;
                }
            }
            
            row.find('.sku-qty-input').val(qtyToFill);
        });
    });

    function filterAndDisplayImports() {
        const searchTerm = $('#importSearchInput').val().toLowerCase();
        const selectedKho = $('#importKhoFilter').val();
        const startDate = $('#importStartDate').val() ? new Date($('#importStartDate').val()) : null;
        const endDate = $('#importEndDate').val() ? new Date($('#importEndDate').val()) : null;
        
        if(startDate) startDate.setHours(0, 0, 0, 0);
        if(endDate) endDate.setHours(23, 59, 59, 999);

        filteredData.import = allData.import.filter(item => {
            const matchesKho = !selectedKho || String(item.MA_KHO) === selectedKho;
            
            const itemDate = parseDate(item.NGAY_CT);
            let matchesDate = true;
            if (startDate && endDate) matchesDate = itemDate >= startDate && itemDate <= endDate;
            else if (startDate) matchesDate = itemDate >= startDate;
            else if (endDate) matchesDate = itemDate <= endDate;
            
            const matchesSearch = !searchTerm || 
                   String(item.SO_CT || '').toLowerCase().includes(searchTerm) ||
                   String(item.MA_HANG || '').toLowerCase().includes(searchTerm) ||
                   String(item.TEN_HANG || '').toLowerCase().includes(searchTerm);
            
            return matchesKho && matchesSearch && matchesDate;
        });

        renderedCounts.import = 0;
        $('#import-list').empty();
        if (filteredData.import.length === 0) {
            $('#import-list').html('<tr><td colspan="12" class="text-center p-6">Không tìm thấy kết quả phù hợp.</td></tr>');
        } else {
            loadMoreImports();
        }
    }

        function filterAndDisplayProducts() {
            const searchTerm = $('#searchInput').val().toLowerCase();
            const selectedKho = $('#khoFilter').val();

            if (!allData.products) {
                filteredData.products = [];
            } else {
                 filteredData.products = allData.products.filter(group => {
                    const matchesKho = !selectedKho || group.MA_KHO === selectedKho;
                    if (!matchesKho) return false;
                    if (!searchTerm) return true;
                    const parentMatch = group.MA_HANG.toLowerCase().includes(searchTerm) ||
                                      group.TEN_HANG.toLowerCase().includes(searchTerm);
                    if (parentMatch) return true;
                    const skuMatch = group.skus.some(sku =>
                        (sku.MA_SKU || '').toLowerCase().includes(searchTerm) ||
                        (sku.TEN_HANG || '').toLowerCase().includes(searchTerm)
                    );
                    if (skuMatch) return true;
                    return false;
                });
            }
            renderProductList(filteredData.products); 
        }

        function loadMoreImports() {
            const data = filteredData.import;
            const start = renderedCounts.import;
            const end = Math.min(start + rowsPerLoad, data.length);
            let rowsHtml = '';
            for (let i = start; i < end; i++) {
                rowsHtml += renderSingleImportRow(data[i]);
            }
            $('#import-list').append(rowsHtml);
            renderedCounts.import = end;
        }

        function displayInventoryReport(data) {
            allData.inventory = data; // Lưu dữ liệu gốc
            $('#inventory-loading-row').hide();
            const listBody = $('#inventory-list');
            const selectedKho = $('#inventoryKhoFilter').val();
            const searchTerm = $('#inventorySearchInput').val().toLowerCase();

            let filteredReport = data.filter(group => {
                const matchesKho = !selectedKho || group.MA_KHO === selectedKho;
                if (!matchesKho) return false;
                if (!searchTerm) return true;

                const parentMatch = group.MA_HANG.toLowerCase().includes(searchTerm) || group.TEN_HANG.toLowerCase().includes(searchTerm);
                if (parentMatch) return true;
                
                const skuMatch = group.skus.some(sku =>
                    (sku.MA_SKU || '').toLowerCase().includes(searchTerm) ||
                    (sku.TEN_HANG || '').toLowerCase().includes(searchTerm)
                );
                return skuMatch;
            });

            if (filteredReport.length === 0) {
                listBody.html('<tr><td colspan="8" class="text-center p-6">Không tìm thấy dữ liệu phù hợp.</td></tr>');
                return;
            }

            renderInventoryList(filteredReport);
        }

        function prepareAndOpenProductModal(productGroup = null) {
            const form = $('#productForm');
            form[0].reset();
            const isEditing = productGroup !== null;

            $('#modalTitle').text(isEditing ? 'Sửa sản phẩm' : 'Thêm sản phẩm');
            form.find('input[name="isEditing"]').val(isEditing);

            if (isEditing) {
                // Populate form with existing data
                form.find('input[name="original_MA_KHO"]').val(productGroup.MA_KHO);
                form.find('input[name="original_MA_HANG"]').val(productGroup.MA_HANG);
                form.find('input[name="MA_KHO"]').val(productGroup.MA_KHO);
                form.find('input[name="MA_HANG"]').val(productGroup.MA_HANG);
                form.find('input[name="TEN_HANG"]').val(productGroup.TEN_HANG);
                form.find('input[name="DVT"]').val(productGroup.DVT);
                form.find('input[name="QUY_CACH"]').val(productGroup.QUY_CACH);
                form.find('input[name="GIA_NHAP"]').val(productGroup.GIA_NHAP);
                form.find('input[name="GIA_BAN"]').val(productGroup.GIA_BAN);
                form.find('textarea[name="GHI_CHU"]').val(productGroup.GHI_CHU);
                form.find('input[name="MAU_SAC"]').val(productGroup.MAU_SAC_GOC);

                const hasVariants = productGroup.skus.length > 1 || (productGroup.skus.length === 1 && productGroup.skus[0].MA_SKU !== productGroup.MA_HANG);
                if (hasVariants) {
                    $('input[name="product_type"][value="variant"]').prop('checked', true).trigger('change');
                     generateVariantRowsForEditing(productGroup.skus);
                } else {
                    $('input[name="product_type"][value="simple"]').prop('checked', true).trigger('change');
                    form.find('input[name="SL_DK"]').val(productGroup.skus[0].SL_DK || 0);
                }
            } else {
                $('input[name="product_type"][value="simple"]').prop('checked', true).trigger('change');
                $('#variant-details-table-wrapper').empty();
            }
            $('#modal-sl-dk, #modal-gia-nhap').trigger('input'); // Recalculate total
            openModal('#productModal');
        }

        $('#page-content').on('click change', function(e) {
            const target = $(e.target);

            // -- Checkbox: Select All --
            if (target.hasClass('select-all-checkbox')) {
                const table = target.data('table');
                const isChecked = target.prop('checked');
                const listId = table === 'products' ? '#product-list' : `#${table}-list`;
                $(listId).find('.row-checkbox').prop('checked', isChecked).trigger('change');
            }

            // -- Checkbox: Row Checkbox --
            if (target.hasClass('row-checkbox')) {
                const table = target.data('table');
                updateDeleteButtonVisibility(table);
            }
            
            // -- Edit Product --
            const editProductBtn = target.closest('.edit-product-btn');
            if (editProductBtn.length) {
                const maKho = editProductBtn.data('ma-kho');
                const maHang = editProductBtn.data('ma-hang');
                const productGroup = allData.products.find(p => p.MA_KHO === maKho && p.MA_HANG === maHang);
                if (productGroup) {
                    prepareAndOpenProductModal(productGroup);
                } else {
                    showToast('Lỗi: Không tìm thấy dữ liệu sản phẩm.');
                }
            }

            // -- Delete Product (single) --
            const deleteProductBtn = target.closest('.delete-product-btn');
            if (deleteProductBtn.length) {
                const maKho = deleteProductBtn.data('ma-kho');
                const maHang = deleteProductBtn.data('ma-hang');
                showConfirmation('Xác nhận xoá sản phẩm', `Bạn có chắc muốn xoá nhóm sản phẩm <strong>${maHang}</strong> (Kho: ${maKho})? Mọi SKU thuộc nhóm này sẽ bị xoá vĩnh viễn.`, () => {
                    google.script.run.withSuccessHandler(handleDeleteSuccess).deleteProducts([{ maKho, maHang }]);
                });
            }

            // -- Delete Products (bulk) --
            const deleteSelectedProductsBtn = target.closest('#delete-selected-products-btn');
            if (deleteSelectedProductsBtn.length) {
                const productsToDelete = [];
                $('#product-list .row-checkbox:checked').each(function() {
                    productsToDelete.push({
                        maKho: $(this).data('ma-kho'),
                        maHang: $(this).data('ma-hang')
                    });
                });
                if (productsToDelete.length > 0) {
                    showConfirmation(`Xác nhận xoá ${productsToDelete.length} sản phẩm`, `Bạn có chắc muốn xoá vĩnh viễn các nhóm sản phẩm đã chọn?`, () => {
                        google.script.run.withSuccessHandler(handleDeleteSuccess).deleteProducts(productsToDelete);
                    });
                }
            }

            // -- Edit Document (Import/Export) --
            const editDocBtn = target.closest('.edit-doc-btn');
            if (editDocBtn.length) {
                const soCt = editDocBtn.data('so-ct');
                const type = editDocBtn.data('type');
                if (type === 'import') {
                    google.script.run.withSuccessHandler(openImportForEditing).getSingleImportDoc(soCt);
                } else {
                    google.script.run.withSuccessHandler(openExportForEditing).getSingleExportDoc(soCt);
                }
            }
            
            // -- Delete Document (single) --
            const deleteDocBtn = target.closest('.delete-doc-btn');
            if(deleteDocBtn.length) {
                const soCt = deleteDocBtn.data('so-ct');
                const type = deleteDocBtn.data('type');
                const sheetName = type === 'import' ? 'DL_NHAP' : 'DL_XUAT';
                const docName = type === 'import' ? 'phiếu nhập' : 'phiếu xuất';
                showConfirmation(`Xác nhận xoá chứng từ`, `Bạn có chắc muốn xoá toàn bộ ${docName} <strong>${soCt}</strong>?`, () => {
                    google.script.run.withSuccessHandler(handleDeleteSuccess).deleteDocuments(sheetName, [soCt]);
                });
            }

            // -- Delete Documents (bulk) --
            const deleteSelectedDocsBtn = target.closest('#delete-selected-imports-btn, #delete-selected-exports-btn');
            if (deleteSelectedDocsBtn.length) {
                const type = deleteSelectedDocsBtn.attr('id').includes('import') ? 'import' : 'export';
                const sheetName = type === 'import' ? 'DL_NHAP' : 'DL_XUAT';
                const docName = type === 'import' ? 'phiếu nhập' : 'phiếu xuất';
                const soCtList = [];
                $(`#${type}-list .row-checkbox:checked`).each(function() {
                    const soCt = $(this).data('so-ct');
                    if(soCt && !soCtList.includes(soCt)) {
                        soCtList.push(soCt);
                    }
                });

                if (soCtList.length > 0) {
                    showConfirmation(`Xác nhận xoá ${soCtList.length} chứng từ`, `Bạn có chắc muốn xoá vĩnh viễn các ${docName} đã chọn?`, () => {
                        google.script.run.withSuccessHandler(handleDeleteSuccess).deleteDocuments(sheetName, soCtList);
                    });
                }
            }
        });

        function updateDeleteButtonVisibility(table) {
            const listId = table === 'products' ? '#product-list' : `#${table}-list`;
            const checkedCount = $(`${listId} .row-checkbox:checked`).length;
            const buttonId = `#delete-selected-${table}${table === 'products' ? '' : 's'}-btn`;
            const deleteBtn = $(buttonId);

            if (checkedCount > 0) {

        $('#login-form').on('submit', function(e) {
            e.preventDefault();
            
            const username = $('#login-username').val().trim();
            const password = $('#login-password').val();
            
            if (!username || !password) {
                showLoginError('Vui lòng nhập đầy đủ thông tin đăng nhập.');
                return;
            }
            
            // Hiển thị loading
            $('#login-btn-text').hide();
            $('#login-btn-loader').show();
            $('#login-btn').prop('disabled', true);
            hideLoginError();
            
            // Gọi API đăng nhập
            google.script.run.withSuccessHandler(function(response) {
                if (response.status === 'success') {
                    // Đăng nhập thành công
                    currentUser = response.data.user;
                    userWarehouses = [];
                    
                    // Lấy thông tin kho
                    google.script.run.withSuccessHandler(function(warehouseResponse) {
                        if (warehouseResponse.status === 'success') {
                            userWarehouses = warehouseResponse.data.warehouses;
                        }
                        
                        // Hiển thị giao diện chính
                        showMainInterface();
                        updateUIByPermissions();
                        updateUserInfoDisplay();
                        
                        // Tải dữ liệu Dashboard
                        switchTab('dashboard');
                        
                    }).getCurrentUserInfo();
                    
                } else {
                    // Đăng nhập thất bại
                    showLoginError(response.message);
                }
                
                // Ẩn loading
                $('#login-btn-text').show();
                $('#login-btn-loader').hide();
                $('#login-btn').prop('disabled', false);
                
            }).login(username, password);
        });
        
        function showLoginError(message) {
            $('#login-error').text(message).removeClass('hidden');

        function hideLoginError() {
            $('#login-error').addClass('hidden');

        function loadSettingsData() {
            if (currentUser && currentUser.ROLE === 'admin') {
                loadUsersList();
                populateUserSelect();
            }
        }
        
        function loadUsersList() {
            $('#users-loading-row').show();
            google.script.run.withSuccessHandler(function(response) {
                $('#users-loading-row').hide();
                if (response.status === 'success') {
                    displayUsersList(response.data);
                } else {
                    showToast('Lỗi: ' + response.message);
                }
            }).getAllUsers();
        }
        
        function displayUsersList(users) {
            const tbody = $('#users-table-body');
            tbody.find('tr:not(#users-loading-row)').remove();
            
            users.forEach(user => {
                const row = `
                    <tr class="border-b hover:bg-gray-50">
                        <td class="p-3">${user.USER}</td>
                        <td class="p-3">${user.USER}</td>
                        <td class="p-3">
                            <span class="px-2 py-1 text-xs rounded-full ${getRoleBadgeClass(user.ROLE)}">
                                ${getRoleDisplayName(user.ROLE)}
                            </span>
                        </td>
                        <td class="p-3">${user.KHO || 'Tất cả kho'}</td>
                        <td class="p-3">
                            <span class="px-2 py-1 text-xs rounded-full ${user.STATUS === 'active' ? 'bg-green-100 text-green-800' : 'bg-red-100 text-red-800'}">
                                ${user.STATUS === 'active' ? 'Hoạt động' : 'Không hoạt động'}
                            </span>
                        </td>
                        <td class="p-3">
                            <button class="edit-user-btn text-blue-600 hover:text-blue-800 text-sm" data-username="${user.USER}">
                                <i class="ph-bold ph-pencil-simple mr-1"></i>Sửa
                            </button>
                        </td>
                    </tr>
                `;
                tbody.append(row);
            });
        }
        
        function getRoleBadgeClass(role) {
            const classes = {
                'admin': 'bg-red-100 text-red-800',
                'accountant': 'bg-blue-100 text-blue-800',
                'warehouse_keeper': 'bg-green-100 text-green-800'
            };
            return classes[role] || 'bg-gray-100 text-gray-800';
        }
        
        function populateUserSelect() {
            google.script.run.withSuccessHandler(function(response) {
                if (response.status === 'success') {
                    const select = $('#user-select');
                    select.find('option:not(:first)').remove();
                    
                    response.data.forEach(user => {
                        select.append(`<option value="${user.EMAIL}">${user.FULL_NAME || user.EMAIL} (${getRoleDisplayName(user.ROLE)})</option>`);
                    });
                }
            }).getAllUsers();
        }
